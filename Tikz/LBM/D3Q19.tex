\documentclass[tikz,border=0pt]{standalone}
\usepackage[portuguese,english]{babel}
%
% Useful packages
\usepackage{bm}
\usepackage[utf8]{inputenc}
\usepackage{pgfplots}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\pgfplotsset{compat=1.18}
\usetikzlibrary{arrows.meta}
\begin{document}
\tdplotsetmaincoords{75}{30}
\tikzset{FixedStealth/.tip = {Stealth[length=6pt, width=4pt]}}
\tikzset{>=FixedStealth}
\begin{tikzpicture}[scale=1, tdplot_main_coords]
    %
    \def\L{2}
    % ---- 1. Draw the grid (edges of the 2×2×2 cube subdivided) ----
    %\path[use as bounding box] (-\L,-\L,-\L) rectangle (\L,\L,\L);
    \tikzset{grid line/.style={gray, dashed}}
    %
    % Lines parallel to x‑axis
    \foreach \y in {-\L,0,\L}
        {
            \foreach \z in {-\L,0,\L}
                {
                    \foreach \xstart in {-\L,0}
                        {
                            \draw[grid line] (\xstart,\y,\z) -- (\xstart+\L,\y,\z);
                        }
                }
        }
    %
    % Lines parallel to y‑axis
    \foreach \x in {-\L,0,\L}
        {
            \foreach \z in {-\L,0,\L}
                {
                    \foreach \ystart in {-\L,0}
                        {
                            \draw[grid line] (\x,\ystart,\z) -- (\x,\ystart+\L,\z);
                        }
                }
        }
    %
    % Lines parallel to z‑axis
    \foreach \x in {-\L,0,\L}
        {
            \foreach \y in {-\L,0,\L}
                {
                    \foreach \zstart in {-\L,0}
                        {
                            \draw[grid line] (\x,\y,\zstart) -- (\x,\y,\zstart+\L);
                        }
                }
        }
    %
    % ---- Coordinate system (placed at a corner for clarity) ----
    %
    \begin{scope}[thick, ->]
        \draw (-\L*1.2,-\L*1.2,-\L*1.2) -- ++(\L*0.8,0,0) node[above] {$x$};
        \draw (-\L*1.2,-\L*1.2,-\L*1.2) -- ++(0,-\L*0.8,0) node[above] {$y$};
        \draw (-\L*1.2,-\L*1.2,-\L*1.2) -- ++(0,0,\L*0.8) node[left] {$z$};
    \end{scope}
    %
    % ---- Draw population lines from centre to every other lattice point ----
    %
    \foreach \x/\y/\z [count=\i] in {
            \L/0/0,  -\L/0/0,  0/\L/0,   0/-\L/0,   0/0/\L,   0/0/-\L,
            \L/\L/0,  \L/-\L/0,  -\L/\L/0,  -\L/-\L/0,
            \L/0/\L,  \L/0/-\L,  -\L/0/\L,  -\L/0/-\L,
            0/\L/\L,  0/\L/-\L,  0/-\L/\L,  0/-\L/-\L
        }
        {
            \draw[->] (0,0,0) -- (\x,\y,\z);
        }
    %
    \node[circle, fill=black, inner sep=1pt] at (0,0,0) {};
    %
\end{tikzpicture}
\end{document}
