\begin{tikzpicture}[
    node distance=64pt, 
    % Define block styles
    process/.style = {rectangle, draw, fill = ApplePalette6!50, text width = 212pt, text centered, rounded corners, minimum height=34pt, font=\small},
    splitprocess/.style = {rectangle, draw, fill = ApplePalette6!50, text width = 160pt, text centered, rounded corners, minimum height=34pt, font=\small},
    decision/.style = {diamond, aspect=2.2, draw, fill = ApplePalette2!50, text width = 75pt, text centered, inner sep = 1pt, font=\small},
    io/.style = {trapezium, trapezium left angle = 75, trapezium right angle = 105, draw, fill = ApplePalette1!50, text width = 198pt, text centered, minimum height = 34pt, font = \small},
    arrow/.style = {thick, ->, >=stealth}
]

    % Nodes
    \node (read) [io] {\textbf{Read from global memory} \\
    $\rho$, $\VelocityVector$, $\MomentsTensor$};
    
    \node (regularize) [process, below of=read] {\textbf{Regularization (reconstruct populations)} \\
    $\widehat{f}_i$ via \cref{eq:Regularization}};
    
    \node (stream) [process, below of=regularize] {\textbf{Shared memory streaming} \\
    $f_{i}$ via \cref{eq:Streaming}};
    
    % --- The Decision Node ---
    \node (is_boundary) [decision, below of=stream] {\textbf{At boundary?}};
    
    % --- The Branch ---
    \node (moments) [splitprocess, below of=is_boundary, xshift=-115pt] {\textbf{Calculate post-streaming moments} \\
    $\rho, \rho\,\VelocityVector, \rho\,\MomentsTensor$ via \cref{eq:Moments}};
    
    \node (bc) [splitprocess, below of=is_boundary, xshift=115pt] {\textbf{Apply boundary conditions} \\
    (IRBC framework)};
    
    % --- Re-join ---
    \node (collide) [process, below of=moments, xshift=115pt] {\textbf{Moment-space collision} \\
    Calculate $\rho^*, \VelocityVector^*, \MomentsTensor^*$ via \cref{eq:Collision}};
    
    \node (write) [io, below of=collide] {\textbf{Write to global memory} \\
    $\rho^*$, $\VelocityVector^*$, $\MomentsTensor^*$};

    % Arrows
    \draw [->, thick] (read) -- (regularize);
    \draw [->, thick] (regularize) -- (stream);
    \draw [->, thick] (stream) -- (is_boundary);
    
    % Forking arrows from the decision diamond
    \draw [->, thick] (is_boundary.west) -| node[above, near start] {No} (moments.north);
    \draw [->, thick] (is_boundary.east) -| node[above, near start] {Yes} (bc.north);
    
    % Joining arrows (perfect symmetrical merge using a fixed coordinate)
    \coordinate (merge) at ([yshift=15pt]collide.north);
    \draw [thick] (moments.south) |- (merge);
    \draw [thick] (bc.south) |- (merge);
    \draw [->, thick] (merge) -- (collide.north);
    
    \draw [->, thick] (collide) -- (write);

    % Feedback loop returning to the start
    \draw [->, thick] (write.east) -- ++(72pt,0) |- (read.east);

\end{tikzpicture}
