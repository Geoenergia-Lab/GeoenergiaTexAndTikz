\documentclass[tikz,border=5pt]{standalone}
\usepackage[portuguese,english]{babel}
\usepackage{bm}
\usepackage[utf8]{inputenc}
\usepackage{pgfplots}
\usepackage{siunitx}
\pgfplotsset{compat=1.18}
\usetikzlibrary{arrows.meta}
%
\begin{document}
\begin{tikzpicture}[scale=1]
%
% Axis limits of the plot
\def\xMin{0.1}
\def\xMax{1000}
\def\yMin{0.01}
\def\yMax{1000}
%
% Add the definition of FLOP to siunitx
\DeclareSIUnit\flop{FLOP}
%
% Generic command to add a GPU roofline plot
% Arg 1: Line colour
% Arg 2: GPU name
% Arg 3: Peak FLOP/s in TB/s
% Arg 4: Memory bandwidth in TB/s
\newcommand{\addRoofline}[4]
{
    \pgfmathsetmacro{\ridgeX}{#3 / #4}
    \draw[#1] (\xMin, \xMin * #4) -- (\ridgeX, #3) -- (\xMax, #3);
    \addlegendimage{line legend, #1}
    \addlegendentry{#2}
}
%
\begin{axis}[
    width = \textwidth,
    height = 0.5\textwidth,
    xmode = log,
    ymode = log,
    clip = true,
    domain = \xMin : \xMax,
    xmin = \xMin,
    xmax = \xMax,
    ymin = \yMin,
    ymax = \yMax,
    xtick = {0.1, 1, 10, 100, 1000},
    ytick = {0.01, 0.1, 1, 10, 100, 1000},
    % ------------------------------------
    axis lines = left,
    xlabel = {Arithmetic intensity [\SI{}{\flop\per\byte}]},
    ylabel = {Performance [\SI{}{\tera\flop\per\second}]},
    xlabel style = {font=\normalsize, below},
    ylabel style = {font=\normalsize, above, rotate=0},
    tick label style = {font=\normalsize},
    grid = both,
    minor tick num = 8,
    minor grid style = {dashed, gray!20},
    samples = 100,
    no marks,
    legend pos = north west,
    legend cell align = left
]
%
\addRoofline{black}{RTX 4090}{82.6}{1.008}
\addRoofline{blue}{RTX A4000}{19.17}{0.448}
\addRoofline{red}{RTX 5090}{104.8}{1.792}
%
\end{axis}
\end{tikzpicture}
\end{document}
