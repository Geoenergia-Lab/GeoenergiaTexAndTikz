\newcommand{\addRoofline}[4]{%
    \pgfmathsetmacro{\ridgeX}{#3 / #4}
    
    % BAKE THE LINE: Force \ridgeX, #3, and #4 into literal numbers
    % \noexpand prevents \draw, \xMin, and \xMax from expanding too early
    \edef\queueDraw{%
        \noexpand\draw[#1, thick] (\noexpand\xMin, \noexpand\xMin * #4) -- (\ridgeX, #3) -- (\noexpand\xMax, #3);
    }%
    \queueDraw % Put the hard-coded math onto the pgfplots to-do list
    
    \addlegendimage{line legend, #1}
    \addlegendentry{#2}
}

% 2. The CSV wrapper
\newcommand{\rooflineFromCSV}[3]{%
  \pgfplotstableread[col sep=comma]{#3}{\tempgpudata}%
  
  \pgfplotstablegetelem{0}{PeakFLOPs}\of{\tempgpudata}%
  \xdef\gpuPeakFLOPs{\pgfplotsretval}%
  
  \pgfplotstablegetelem{0}{Bandwidth}\of{\tempgpudata}%
  \xdef\gpuBandwidth{\pgfplotsretval}%
  
  % BAKE THE NUMBERS: Force variables into literal numbers before calling
  \edef\executeRoofline{%
      \noexpand\addRoofline{\unexpanded{#1}}{\unexpanded{#2}}{\gpuPeakFLOPs}{\gpuBandwidth}%
  }%
  \executeRoofline % Execute the baked command
  
  % Garbage collection is now 100% safe
  \global\let\gpuPeakFLOPs\undefined
  \global\let\gpuBandwidth\undefined
}

\newcommand{\addPerformancePoint}[2]
{
    \draw plot[
        only marks,
        mark = o,
        mark size = 3pt,
        mark options = {color = #1}
    ] coordinates {(#2)};
    \draw plot[
        only marks,
        mark = *,
        mark size = 1pt,
        mark options = {color = #1}
    ] coordinates {(#2)};
}
\newcommand{\perfPointFromCSV}[2]{%
  % #1 : Color
  % #2 : Path to CSV file

  % 1. Load the CSV into a local macro
  \pgfplotstableread[col sep=comma]{#2}{\tempgpudata}%
  
  % 2. Extract Arithmetic Intensity (AI)
  \pgfplotstablegetelem{0}{AI}\of{\tempgpudata}%
  \xdef\gpuAI{\pgfplotsretval}%
  
  % 3. Extract Performance (P)
  \pgfplotstablegetelem{0}{P}\of{\tempgpudata}%
  \xdef\gpuPerf{\pgfplotsretval}%
  
  % 4. BAKE THE NUMBERS: Force variables into literal text.
  % This safely injects "3.02, 2.72" as argument #2 into your macro
  \edef\executePerfPoint{%
      \noexpand\addPerformancePoint{\unexpanded{#1}}{\gpuAI, \gpuPerf}%
  }%
  \executePerfPoint
  
  % 5. Garbage collection
  \global\let\gpuAI\undefined
  \global\let\gpuPerf\undefined
}
