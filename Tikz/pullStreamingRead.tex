    % Grid lines
    \foreach \i in {0,...,3}
    {
        \draw[lightgray!50, semithick] (-0.25, \i) -- (3.25, \i);
        \draw[lightgray!50, semithick] (\i, -0.25) -- (\i, 3.25);
    }
    
    % Background D2Q9 arrows
    \foreach \x in {0,...,2}
    {
        \foreach \y in {0,...,2}
        {
            \draw[bgarrow] (\x, \y+0.5) -- (\x+1, \y+0.5);
            \draw[bgarrow] (\x+0.5, \y) -- (\x+0.5, \y+1);
            \draw[bgarrow] (\x, \y) -- (\x+1, \y+1);
            \draw[bgarrow] (\x+1, \y) -- (\x, \y+1);
        }
    }
    
    % Foreground pull arrows (read)
    \draw[fgarrow] (0.5,1.5) -- (1,1.5);
    \draw[fgarrow] (2.5,1.5) -- (2,1.5);
    \draw[fgarrow] (1.5,0.5) -- (1.5,1);

    \draw[fgarrow] (2.5,2.5) -- (2,2);
    
    % Nodes
    \foreach \x in {0,...,3}
    {
        \foreach \y in {0,...,3}
        {
            \fill[lightgray] (\x,\y) circle (1.5pt);
        }
    }
