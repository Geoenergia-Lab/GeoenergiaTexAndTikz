\newcommand{\plotUyN}[3]{%
  \addplot [#2, #3] table [
    ignore chars={\"},
    x expr=(\thisrow{Points:0} - 0.5) * 2,
    y expr=\thisrow{UMean_y} / (3 * ((#1 - 1) / 10000)),
    col sep=comma
  ] {Data/Mesh_#1/uyNoDuplicates.csv};
}
